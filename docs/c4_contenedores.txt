        soporte = person "Soporte y administrador" "Resuelve pagos en revision, reembolsos, bloqueos y liquidaciones."

            broker = container "Message broker" "Transporta los eventos de negocio entre servicios." "Message broker" "Broker"

        gateway -> identidad "Enruta solicitudes" "HTTP/JSON"
        gateway -> catalogo "Enruta solicitudes" "HTTP/JSON"
        gateway -> inventario "Enruta solicitudes" "HTTP/JSON"
        gateway -> reservas "Enruta solicitudes" "HTTP/JSON"

        gateway -> auditoria "Enruta solicitudes" "HTTP/JSON"

        identidad -> identidadDb "SQL"

        inventario -> inventarioDb "SQL"
        reservas -> reservasDb "SQL"
        entradas -> entradasDb "SQL"
